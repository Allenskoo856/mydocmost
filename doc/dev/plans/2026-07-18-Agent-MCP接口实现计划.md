# Agent MCP 接口实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 mydocmost 后端实现一套基于 MCP 2024-11 SSE 协议的 Agent 接口，支持跨 Workspace 的空间/页面树/Markdown 读写操作。

**Architecture:** 在 NestJS 中新增 `core/mcp` 模块；通过环境变量 `MCP_API_TOKEN` 认证；每个 Workspace 拥有一个系统 Agent 用户；复用现有 `SpaceService`、`PageService`、`ImportService`、`ExportService` 完成业务逻辑；Markdown 与 ProseMirror 之间的转换复用已有工具函数。

**Tech Stack:** NestJS 11、Fastify、MCP SDK 2024-11、Kysely、PostgreSQL、class-validator、Jest。

## Global Constraints

- 所有新增代码使用 TypeScript，遵循项目 ESLint 9 flat config。
- 服务端代码位于 `apps/server/src/`。
- 数据库类型 `apps/server/src/database/types/db.d.ts` 禁止手改，只能由 `migration:codegen` 生成。
- 修改后端后必须运行 `pnpm server:build` 和相关 Jest 测试。
- 子目录部署：`BASE_PATH` 必须以 `/` 开头、不以 `/` 结尾；MCP 端点需拼接在 `BASE_PATH` 之后。
- 内网部署约束：不得引入外网调用；保持离线可用。
- 环境变量校验使用 `class-validator`，新增变量在 `EnvironmentVariables` 中声明。
- 代码注释与标识符使用英文；面向用户的文档使用中文。

- [ ] **Step 1: 编写 Guard**

```ts
import {
  CanActivate,
  ExecutionContext,
  Injectable,
  UnauthorizedException,
} from '@nestjs/common';
import { FastifyRequest } from 'fastify';
import { EnvironmentService } from '../../integrations/environment/environment.service';

@Injectable()
export class McpAuthGuard implements CanActivate {
  constructor(private readonly environmentService: EnvironmentService) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const expectedToken = this.environmentService.getMcpApiToken();
    if (!expectedToken) {
      throw new UnauthorizedException('MCP is not enabled');
    }

    const request = context.switchToHttp().getRequest<FastifyRequest>();
    const authHeader = request.headers.authorization || '';
    const token = authHeader.startsWith('Bearer ') ? authHeader.slice(7) : '';

    if (!token) {
      throw new UnauthorizedException('Missing MCP API token');
    }

    if (!this.timingSafeEqual(token, expectedToken)) {
      throw new UnauthorizedException('Invalid MCP API token');
    }

    return true;
  }

  private timingSafeEqual(a: string, b: string): boolean {
    if (a.length !== b.length) {
      // 为避免直接返回，继续做一次比较以模糊长度差异
      // 实际生产仍建议固定长度 token
      const bufA = Buffer.from(a);
      const bufB = Buffer.alloc(bufA.length, 0);
      return crypto.timingSafeEqual(bufA, bufB) && false;
    }
    return crypto.timingSafeEqual(Buffer.from(a), Buffer.from(b));
  }
}
```

> 顶部需 `import * as crypto from 'crypto';`

```ts
import { Injectable, Logger, OnModuleInit } from '@nestjs/common';
import { UserRepo } from '@docmost/db/repos/user/user.repo';
import { WorkspaceRepo } from '@docmost/db/repos/workspace/workspace.repo';
import { EnvironmentService } from '../../integrations/environment/environment.service';
import { User } from '@docmost/db/types/entity.types';
import { UserRole } from '../../common/helpers/types/permission';
import { v7 as uuid7 } from 'uuid';

@Injectable()
export class McpAgentUserService implements OnModuleInit {
  private readonly logger = new Logger(McpAgentUserService.name);

  constructor(
    private readonly userRepo: UserRepo,
    private readonly workspaceRepo: WorkspaceRepo,
    private readonly environmentService: EnvironmentService,
  ) {}

  async onModuleInit(): Promise<void> {
    if (!this.environmentService.getMcpApiToken()) {
      this.logger.log('MCP is not enabled, skip agent user initialization');
      return;
    }

    const workspaces = await this.workspaceRepo.findAll();
    for (const workspace of workspaces) {
      try {
        await this.findOrCreateForWorkspace(workspace.id);
        this.logger.log(`MCP agent user ensured for workspace ${workspace.id}`);
      } catch (err) {
        this.logger.error(
          `Failed to ensure MCP agent user for workspace ${workspace.id}`,
          err,
        );
      }
    }
  }

  async findOrCreateForWorkspace(workspaceId: string): Promise<User> {
    const email = this.environmentService.getMcpAgentUserEmail();
    const existing = await this.userRepo.findByEmail(email, workspaceId);

    if (existing && !existing.deletedAt && !existing.deactivatedAt) {
      return existing;
    }

    if (existing && (existing.deletedAt || existing.deactivatedAt)) {
      throw new Error(`MCP agent user is disabled in workspace ${workspaceId}`);
    }

    const workspace = await this.workspaceRepo.findById(workspaceId);
    if (!workspace) {
      throw new Error(`Workspace ${workspaceId} not found`);
    }

    return this.userRepo.insertUser({
      id: uuid7(),
      name: 'Docmost Agent',
      email,
      role: UserRole.ADMIN,
      workspaceId,
      password: uuid7(), // random, prevents normal login
    });
  }
}
```

> 注：`UserRepo.findByEmail` 和 `UserRepo.insertUser` 签名需与项目实际一致，可能需调整字段名。

- [ ] **Step 2: 编写 `McpController`**

```ts
import { Controller, Get, Logger, Post, Req, Res, UseGuards } from '@nestjs/common';
import { FastifyReply, FastifyRequest } from 'fastify';
import { McpAuthGuard } from './mcp-auth.guard';
import { McpServerService } from './mcp-server.service';
import { SSEServerTransport } from '@modelcontextprotocol/sdk/server/sse.js';

@UseGuards(McpAuthGuard)
@Controller('mcp')
export class McpController {
  private readonly logger = new Logger(McpController.name);
  private readonly transports = new Map<string, SSEServerTransport>();

  constructor(private readonly mcpServerService: McpServerService) {}

  @Get('sse')
  async sse(@Req() req: FastifyRequest, @Res() res: FastifyReply) {
    const transport = new SSEServerTransport('/mcp/messages', res.raw);
    this.transports.set(transport.sessionId, transport);

    transport.onclose = () => {
      this.transports.delete(transport.sessionId);
      this.logger.log(`MCP session ${transport.sessionId} closed`);
    };

    const server = this.mcpServerService.createServer();
    await server.connect(transport);
    this.logger.log(`MCP session ${transport.sessionId} connected`);
  }

  @Post('messages')
  async messages(
    @Req() req: FastifyRequest,
    @Res() res: FastifyReply,
  ) {
    const sessionId = req.query?.sessionId as string;
    const transport = this.transports.get(sessionId);
    if (!transport) {
      res.status(404).send('Session not found');
      return;
    }
    await transport.handlePostMessage(req.raw, res.raw);
  }
}
```

```ts
import { Server } from '@modelcontextprotocol/sdk/server/index.js';
import { CallToolRequestSchema } from '@modelcontextprotocol/sdk/server/server.js';
import { McpContextService } from '../mcp-context.service';
import { WorkspaceService } from '../../workspace/services/workspace.service';
import { SpaceService } from '../../space/services/space.service';
import { McpCreateSpaceDto } from '../dto/create-space.dto';
import { validate } from 'class-validator';
import { plainToInstance } from 'class-transformer';

export function registerWorkspaceTools(
  server: Server,
  contextService: McpContextService,
  workspaceService: WorkspaceService,
  spaceService: SpaceService,
) {
  server.setRequestHandler(CallToolRequestSchema, async (request) => {
    const { name, arguments: args } = request.params;

    if (name === 'list_workspaces') {
      const workspaces = await workspaceService.findAll();
      return {
        content: [
          {
            type: 'text',
            text: JSON.stringify({ workspaces }, null, 2),
          },
        ],
      };
    }

    if (name === 'create_space') {
      const dto = plainToInstance(McpCreateSpaceDto, args);
      const errors = await validate(dto);
      if (errors.length) {
        return formatError('VALIDATION_ERROR', errors);
      }

      const { user, workspace } = await contextService.load(dto.workspaceId);
      const space = await spaceService.createSpace(user, workspace.id, dto);
      return {
        content: [
          { type: 'text', text: JSON.stringify(space, null, 2) },
        ],
      };
    }

    // fallback
    return null;
  });
}
```

> 注：`workspaceService.findAll()` 如果不存在，需在 `WorkspaceService` 中补充。

```ts
import {
  htmlToJson,
  jsonToText,
} from '../../../collaboration/collaboration.util';
import { markdownToHtml } from '@docmost/editor-ext';
import { createYdocFromJson } from '../../../common/helpers/prosemirror/utils';

export async function markdownToPageContent(markdown: string): Promise<{
  title?: string;
  content: any;
  textContent: string;
  ydoc: Buffer;
}> {
  const html = await markdownToHtml(markdown);
  const prosemirrorJson = await htmlToJson(html);

  // extract title from first h1
  let title: string | undefined;
  const firstHeading = prosemirrorJson.content?.find((node: any) => node.type === 'heading' && node.attrs?.level === 1);
  if (firstHeading?.content?.[0]?.text) {
    title = firstHeading.content[0].text;
    prosemirrorJson.content = prosemirrorJson.content.filter(
      (node: any) => !(node.type === 'heading' && node.attrs?.level === 1 && node === firstHeading),
    );
  }

  return {
    title,
    content: prosemirrorJson,
    textContent: jsonToText(prosemirrorJson),
    ydoc: createYdocFromJson(prosemirrorJson),
  };
}
```

```ts
export function registerPageTools(
  server: Server,
  contextService: McpContextService,
  pageService: PageService,
  pageRepo: PageRepo,
) {
  server.setRequestHandler(CallToolRequestSchema, async (request) => {
    const { name, arguments: args } = request.params;

    if (name === 'create_page') {
      const dto = plainToInstance(McpCreatePageDto, args);
      const errors = await validate(dto);
      if (errors.length) return formatError('VALIDATION_ERROR', errors);

      const { user, workspace } = await contextService.load(dto.workspaceId);
      const pageContent = await markdownToPageContent(dto.content);
      const createPageDto: CreatePageDto = {
        spaceId: dto.spaceId,
        parentPageId: dto.parentPageId,
        title: dto.title ?? pageContent.title ?? 'Untitled',
        icon: dto.icon,
      };

      const page = await pageService.create(user.id, workspace.id, createPageDto);
      // 更新 content/textContent/ydoc
      await pageRepo.updatePage(
        {
          content: pageContent.content,
          textContent: pageContent.textContent,
          ydoc: pageContent.ydoc,
          lastUpdatedById: user.id,
          updatedAt: new Date(),
        },
        page.id,
      );

      return { content: [{ type: 'text', text: JSON.stringify(page, null, 2) }] };
    }

    // update_page, move_page 类似

    return null;
  });
}
```

```ts
export function registerExportTools(
  server: Server,
  contextService: McpContextService,
  pageRepo: PageRepo,
  exportService: ExportService,
) {
  server.setRequestHandler(CallToolRequestSchema, async (request) => {
    const { name, arguments: args } = request.params;

    if (name === 'get_page_markdown') {
      const dto = plainToInstance(McpGetPageMarkdownDto, args);
      const errors = await validate(dto);
      if (errors.length) return formatError('VALIDATION_ERROR', errors);

      await contextService.load(dto.workspaceId);
      const page = await pageRepo.findById(dto.pageId, { includeContent: true });
      if (!page || page.workspaceId !== dto.workspaceId) {
        return formatError('PAGE_NOT_FOUND', 'Page not found');
      }

      const markdown = await exportService.exportPage('markdown', page, true);
      return {
        content: [
          {
            type: 'text',
            text: JSON.stringify(
              {
                pageId: page.id,
                title: page.title,
                markdown,
                updatedAt: page.updatedAt,
                spaceId: page.spaceId,
              },
              null,
              2,
            ),
          },
        ],
      };
    }

    return null;
  });
}
```
